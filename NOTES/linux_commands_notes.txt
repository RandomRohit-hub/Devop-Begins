Linux/Bash Commands and Their Uses

cd /
- Move to root directory

ls
- List files and folders

ls /
- List contents of root directory

chmod +x file.sh
- Make script executable

./script.sh
- Run shell script

cat file
- Display file contents

cat /var/run/httpd/httpd.pid
- Show Apache PID

echo $?
- Show exit status of previous command

systemctl start httpd
- Start Apache service

systemctl stop httpd
- Stop Apache service

systemctl restart httpd
- Restart Apache service

systemctl status httpd
- Check Apache status

systemctl enable httpd
- Enable Apache on boot

systemctl is-active httpd
- Check if service is running

yum install httpd -y
- Install Apache package

ps -ef | grep httpd
- Find Apache process

rpm -qa | grep httpd
- Check installed Apache package

sudo -i
- Switch to root user

vagrant ssh scriptbox
- SSH into Vagrant VM

pwd
- Show current directory

mkdir name
- Create directory

rm file
- Remove file

rm -r folder
- Remove directory recursively

clear
- Clear terminal screen

date
- Show current date/time

sleep 3
- Pause script for 3 seconds

read VAR
- Take user input

read -p "msg" VAR
- Input with prompt

if [ condition ]
- Conditional check

then
- Start if block

else
- Alternate block

fi
- End if block

-eq
- Equal comparison

-gt
- Greater than comparison

grep word file
- Search text pattern

#!/bin/bash
- Shebang for Bash interpreter
